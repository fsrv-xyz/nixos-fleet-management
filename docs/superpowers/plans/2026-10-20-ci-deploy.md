# CI-triggered deploys Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A CI pipeline assigns a system closure to an agent through the fleet servers' mTLS port 8443 with a deploy certificate, optionally waiting for the outcome.

**Architecture:** The fleet CA issues deploy certificates (`OU=deploy`) and keeps them under `/ca/deployers/<name>`. A new gRPC service `Deployments` on the existing agent listener authorizes only the certificate on file, writes `/agents/<id>/desired` and, on request, streams the agent's progress from `/agents/<id>/deploy`. A new binary `fleet-deploy` is the CI client.

**Tech Stack:** Go 1.27, gRPC, etcd clientv3, protobuf (`go generate ./api`), embedded etcd in tests.

**Spec:** `docs/superpowers/specs/2026-10-20-ci-deploy-design.md`

## Global Constraints

- No new Go dependencies; `nix/package.nix` `vendorHash` stays unchanged.
- Deploy certificates: `CN=<name>, OU=deploy`, EKU ClientAuth, valid `agentCertValidity` (825 days), stored under `/ca/deployers/<name>`.
- `Connect` and `RenewCertificate` reject `OU=deploy` with `PermissionDenied`.
- `Deploy` accepts only an `OU=deploy` certificate equal to the one on file, else `PermissionDenied`; invalid input `InvalidArgument`; certname without `/ca/signed/` entry `NotFound`.
- The client's deadline bounds the wait; the server has no timeout of its own.
- `fleet-deploy` exits 0 on DONE or after assigning without `-wait`, 1 otherwise.
- `fleet-server deploy` (etcd) stays.
- Code style: comments only for a non-obvious why; match surrounding code.
- Run Go through the dev shell: `nix develop -c go …`. CI runs `gofmt`, `go vet` and `go test -race ./...`.
- Work on branch `feat/ci-deploy`; `main` is protected. Commit messages carry no tool attribution.

## Review Focus

- Redeploying a store path whose DONE is still stored from an earlier run: the wait must not end on the stale DONE → watch starts after the desired write (Task 2 test pre-stores a DONE).
- Timeout while waiting: client fails with `DeadlineExceeded`, desired path stays set (Task 2 test).
- Reissuing a deploy certificate for the same name: the old certificate stops working (Task 1 test).
- Malformed certname (`../agent`) or store path (`/nix/store/x; reboot`): rejected before touching etcd, desired unchanged (Task 2 test).
- Progress for another store path arriving during the wait: ignored (Task 2 test).

// IssueDeployer creates a key and a deploy certificate for name and keeps the
// certificate on file, replacing and so revoking an earlier one.
func (c *CA) IssueDeployer(ctx context.Context, name string) (certPEM, keyPEM []byte, err error) {
	if !certnamePattern.MatchString(name) {
		return nil, nil, fmt.Errorf("invalid name %q", name)
	}
	public, private, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		return nil, nil, err
	}
	now := time.Now()
	der, err := c.sign(&x509.Certificate{
		Subject:     pkix.Name{CommonName: name, OrganizationalUnit: []string{deployerOU}},
		NotBefore:   now,
		NotAfter:    now.Add(agentCertValidity),
		KeyUsage:    x509.KeyUsageDigitalSignature,
		ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageClientAuth},
	}, public)
	if err != nil {
		return nil, nil, err
	}
	keyDER, err := x509.MarshalPKCS8PrivateKey(private)
	if err != nil {
		return nil, nil, err
	}
	certPEM = pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})
	if _, err := c.etcd.Put(ctx, deployersPrefix+name, string(certPEM)); err != nil {
		return nil, nil, err
	}
	return certPEM, pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: keyDER}), nil
}

func (c *CA) RevokeDeployer(ctx context.Context, name string) error {
	resp, err := c.etcd.Delete(ctx, deployersPrefix+name)
	if err != nil {
		return err
	}
	if resp.Deleted == 0 {
		return fmt.Errorf("no deploy certificate for %q", name)
	}
	return nil
}

// authorizeDeployer accepts only the deploy certificate on file for its name, so
// reissuing or revoking takes effect on the next call.
func (c *CA) authorizeDeployer(ctx context.Context, cert *x509.Certificate) error {
	if !isDeployer(cert) {
		return status.Error(codes.PermissionDenied, "not a deploy certificate")
	}
	name := cert.Subject.CommonName
	resp, err := c.etcd.Get(ctx, deployersPrefix+name)
	if err != nil {
		return status.Errorf(codes.Unavailable, "read deploy certificate: %v", err)
	}
	if len(resp.Kvs) == 0 {
		return status.Errorf(codes.PermissionDenied, "no deploy certificate on file for %s", name)
	}
	onFile, err := parseCertificate(resp.Kvs[0].Value)
	if err != nil {
		return err
	}
	if !onFile.Equal(cert) {
		return status.Errorf(codes.PermissionDenied, "deploy certificate for %s was replaced", name)
	}
	return nil
}

// isDeployer relies on issue setting only the CN of agent certificates, so
// agents can never obtain deployerOU.
func isDeployer(cert *x509.Certificate) bool {
	return slices.Contains(cert.Subject.OrganizationalUnit, deployerOU)
}

	list := len(args) == 1 && args[0] == "list"
	sign := len(args) > 1 && args[0] == "sign"
	deployer := len(args) == 2 && args[0] == "deployer"

func validateDeploy(certname, storePath string) error {
	if !certnamePattern.MatchString(certname) {
		return fmt.Errorf("invalid certname %q", certname)
	}
	if !storePathPattern.MatchString(storePath) {
		return fmt.Errorf("invalid store path %q", storePath)
	}
	return nil
}

// SetDesired makes the agent with certname switch to storePath and returns the
// etcd revision of that change.
func SetDesired(ctx context.Context, etcd *clientv3.Client, certname, storePath string) (int64, error) {
	if err := validateDeploy(certname, storePath); err != nil {
		return 0, err
	}
	resp, err := etcd.Put(ctx, agentsPrefix+certname+"/desired", storePath)
	if err != nil {
		return 0, err
	}
	return resp.Header.Revision, nil
}

package fleet

import (
	"context"
	"crypto/tls"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"log"

	clientv3 "go.etcd.io/etcd/client/v3"
	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/credentials"
	"google.golang.org/grpc/status"

	"nixos-fleet-management/api"
)

type Deployments struct {
	api.UnimplementedDeploymentsServer
	ca   *CA
	etcd *clientv3.Client
}

func (d *Deployments) Deploy(req *api.DeployRequest, stream grpc.ServerStreamingServer[api.DeployProgress]) error {
	ctx := stream.Context()
	cert, err := clientCertificate(ctx)
	if err != nil {
		return err
	}
	if err := d.ca.authorizeDeployer(ctx, cert); err != nil {
		return err
	}
	if err := validateDeploy(req.Certname, req.StorePath); err != nil {
		return status.Error(codes.InvalidArgument, err.Error())
	}
	signed, err := d.etcd.Get(ctx, signedPrefix+req.Certname, clientv3.WithCountOnly())
	if err != nil {
		return status.Errorf(codes.Unavailable, "look up %s: %v", req.Certname, err)
	}
	if signed.Count == 0 {
		return status.Errorf(codes.NotFound, "no agent %s", req.Certname)
	}
	revision, err := SetDesired(ctx, d.etcd, req.Certname, req.StorePath)
	if err != nil {
		return status.Errorf(codes.Unavailable, "set desired state: %v", err)
	}
	log.Printf("deploy %s %s by %s", req.Certname, req.StorePath, cert.Subject.CommonName)
	if !req.Wait {
		return nil
	}
	return d.follow(ctx, stream, req.Certname, req.StorePath, revision)
}

// follow watches from just after the desired change, so progress left from an
// earlier deployment of the same store path does not count.
func (d *Deployments) follow(ctx context.Context, stream grpc.ServerStreamingServer[api.DeployProgress], certname, storePath string, revision int64) error {
	for wr := range d.etcd.Watch(ctx, agentsPrefix+certname+"/deploy", clientv3.WithRev(revision+1)) {
		if err := wr.Err(); err != nil {
			return status.Errorf(codes.Unavailable, "watch progress: %v", err)
		}
		for _, ev := range wr.Events {
			if ev.Type == clientv3.EventTypeDelete {
				continue
			}
			var deploy DeployStatus
			if err := json.Unmarshal(ev.Kv.Value, &deploy); err != nil {
				return err
			}
			if deploy.StorePath != storePath {
				continue
			}
			phase := api.DeployProgress_Phase(api.DeployProgress_Phase_value[deploy.Phase])
			if err := stream.Send(&api.DeployProgress{StorePath: storePath, Phase: phase, Message: deploy.Message}); err != nil {
				return err
			}
			if phase == api.DeployProgress_PHASE_DONE || phase == api.DeployProgress_PHASE_FAILED {
				return nil
			}
		}
	}
	if err := ctx.Err(); err != nil {
		return status.FromContextError(err).Err()
	}
	return status.Error(codes.Unavailable, "progress watch closed")
}

// RequestDeploy assigns storePath to certname through one of addrs and, with
// wait, reports progress until the deployment ends; ctx bounds both.
func RequestDeploy(ctx context.Context, addrs []string, caFile, certFile, keyFile, certname, storePath string, wait bool, report func(*api.DeployProgress)) error {
	pool, err := loadPool(caFile)
	if err != nil {
		return err
	}
	pair, err := tls.LoadX509KeyPair(certFile, keyFile)
	if err != nil {
		return err
	}
	conn, err := dial(addrs, credentials.NewTLS(&tls.Config{MinVersion: tls.VersionTLS13, RootCAs: pool, Certificates: []tls.Certificate{pair}}))
	if err != nil {
		return err
	}
	defer conn.Close()
	stream, err := api.NewDeploymentsClient(conn).Deploy(ctx, &api.DeployRequest{Certname: certname, StorePath: storePath, Wait: wait}, grpc.WaitForReady(true))
	if err != nil {
		return err
	}
	for {
		progress, err := stream.Recv()
		if errors.Is(err, io.EOF) {
			if wait {
				return errors.New("server ended the stream before the deployment finished")
			}
			return nil
		}
		if err != nil {
			return err
		}
		report(progress)
		switch progress.Phase {
		case api.DeployProgress_PHASE_DONE:
			return nil
		case api.DeployProgress_PHASE_FAILED:
			return fmt.Errorf("deployment of %s failed: %s", storePath, progress.Message)
		}
	}
}

package main

import (
	"context"
	"flag"
	"fmt"
	"log"
	"net"
	"os"
	"time"

	"nixos-fleet-management/api"
	"nixos-fleet-management/fleet"
)

const connectTimeout = 30 * time.Second

type config struct {
	// Servers holds host:port addresses or a single "srv:<name>".
	Servers []string `json:"servers"`
	CA      string   `json:"ca"`
	// Cert and Key may name the same file holding both.
	Cert string `json:"cert"`
	Key  string `json:"key"`
}

func main() {
	path := flag.String("config", "", "JSON config file; without it, development defaults apply")
	wait := flag.Duration("wait", 0, "wait this long for the agent to finish the deployment")
	flag.Usage = func() {
		fmt.Fprintln(flag.CommandLine.Output(), "usage: fleet-deploy [-config file] [-wait duration] <certname> <store path>")
		flag.PrintDefaults()
	}
	flag.Parse()
	if flag.NArg() != 2 {
		flag.Usage()
		os.Exit(2)
	}

	cfg := config{
		Servers: []string{"localhost:8443"},
		CA:      "ca/ca.crt",
		Cert:    "deploy.pem",
		Key:     "deploy.pem",
	}
	if err := fleet.LoadConfig(*path, &cfg); err != nil {
		log.Fatal(err)
	}

	timeout := connectTimeout
	if *wait > 0 {
		timeout = *wait
	}
	ctx, cancel := context.WithTimeout(context.Background(), timeout)
	defer cancel()
	servers, err := fleet.ResolveServers(ctx, net.DefaultResolver, cfg.Servers)
	if err != nil {
		log.Fatal(err)
	}
	err = fleet.RequestDeploy(ctx, servers, cfg.CA, cfg.Cert, cfg.Key, flag.Arg(0), flag.Arg(1), *wait > 0,
		func(p *api.DeployProgress) { fmt.Println(p.Phase, p.Message) })
	if err != nil {
		log.Fatal(err)
	}
}
